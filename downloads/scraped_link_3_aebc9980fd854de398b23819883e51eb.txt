Decommissioning of eSRS.gov

Service Contract Reporting for FY25 Starts Oct 3
The service contract reporting (SCR) period for fiscal year 2025 starts on Oct 3, 2025 and closes on Jan 31, 2026. The SCR feature in SAM.gov allows contractors to submit annual reports on service contract activity, as required under FAR 52.204-14 and FAR 52.204-15.

var player;
          var ccEnabled = true;
          var isPlaying = false;

          var tag = document.createElement('script');
          tag.src = "https://www.youtube.com/iframe_api";
          var firstScriptTag = document.getElementsByTagName('script')[0];
          firstScriptTag.parentNode.insertBefore(tag, firstScriptTag);

          function onYouTubeIframeAPIReady() {
              player = new YT.Player('player', {
                  height: '360',
                  width: '640',
                  videoId: '4RSHjczdxq8', 
                  playerVars: { 
                      'playsinline': 1,
                      'controls': 0,
                      'cc_load_policy': 1
                  },
                  events: {
                      'onReady': onPlayerReady,
                      'onStateChange': onPlayerStateChange
                  }
              });
          }
          // The API will call this function when the video player is ready.
          function onPlayerReady(event) {
              var restartButton = document.getElementById('restart');
              var rewindButton = document.getElementById('rewind');
              var playButton = document.getElementById('play');
              var pauseButton = document.getElementById('pause');
              var fastForwardButton = document.getElementById('fast-forward');
              var volumeControl = document.getElementById('volume');
              var ccButton = document.getElementById('cc');
              var fullscreenButton = document.getElementById('fullscreen');
              var progressBar = document.getElementById('progress');
              var currentTimeDisplay = document.getElementById('current-time');
              var progressBarContainer = document.getElementById('progress-bar');
              var progressTooltip = document.getElementById('progress-tooltip');
              var videoContainer = document.querySelector('.px-video-container');
              var controls = document.querySelector('.px-video-controls');
              var muteButton = document.getElementById('mute');
              muteButton.addEventListener('click', function() {
                if(player.isMuted()) {
                    player.unMute();
                }
                else {
                    player.mute();
                }
              });
              fullscreenButton.addEventListener('click', function() {
                  toggleFullScreen();
                  if (document.fullscreenElement) {
                      videoContainer.classList.add('fullscreen');
                  } else {
                      videoContainer.classList.remove('fullscreen');
                  }
              });
              restartButton.addEventListener('click', function() {
                  player.seekTo(0, true);
                  player.playVideo();
                  togglePlayPauseButton(true);
              });
              rewindButton.addEventListener('click', function() {
                  var currentTime = player.getCurrentTime();
                  player.seekTo(currentTime - 20, true);
              });
              playButton.addEventListener('click', function() {
                  player.playVideo();
                  togglePlayPauseButton(true);
              });
              pauseButton.addEventListener('click', function() {
                  player.pauseVideo();
                  togglePlayPauseButton(false);
              });
              fastForwardButton.addEventListener('click', function() { 
                  var currentTime = player.getCurrentTime();
                  player.seekTo(currentTime + 20, true);
              });
              volumeControl.addEventListener('input', function() {
                  var volume = volumeControl.value * 10;
                  player.setVolume(volume);
              });
              ccButton.addEventListener('click', function() {
                  if (ccEnabled) {
                      player.unloadModule('captions');
                  } else {
                      player.loadModule('captions');
                  }
                  ccEnabled = !ccEnabled;
              });
              document.addEventListener('fullscreenchange', function() {
                  if (document.fullscreenElement) {
                      controls.classList.add('js-fullscreen-controls');
                      videoContainer.classList.add('fullscreen');
                  } else {
                      controls.classList.remove('js-fullscreen-controls');
                      videoContainer.classList.remove('fullscreen');
                  }
              });
              progressBarContainer.addEventListener('mousemove', function(e) {
                  var rect = progressBarContainer.getBoundingClientRect();
                  var mouseX = e.clientX - rect.left;
                  var duration = player.getDuration();
                  var hoverTime = (mouseX / rect.width) * duration;
                  var minutes = Math.floor(hoverTime / 60);
                  var seconds = Math.floor(hoverTime % 60).toString().padStart(2, '0');
                  progressTooltip.innerHTML = minutes + ':' + seconds;
                  progressTooltip.style.left = mouseX + 'px';
                  progressTooltip.style.display = 'block';
              });
              progressBarContainer.addEventListener('mouseleave', function() {
                  progressTooltip.style.display = 'none';
              });
              progressBarContainer.addEventListener('click', function(e) {
                  var rect = progressBarContainer.getBoundingClientRect();
                  var clickX = e.clientX - rect.left;
                  var duration = player.getDuration();
                  var clickTime = (clickX / rect.width) * duration;
                  player.seekTo(clickTime, true);
              });
              setInterval(function() {
                  updateProgressBar();
                  updateCurrentTimeDisplay();
              }, 1000);
          }
          function onPlayerStateChange(event) {
              if (event.data == YT.PlayerState.PLAYING) {
                  togglePlayPauseButton(true);
              } else if (event.data == YT.PlayerState.PAUSED) {
                  togglePlayPauseButton(false);
              }
          }
          function updateProgressBar() {
              var progressBar = document.getElementById('progress');
              var currentTime = player.getCurrentTime();
              var duration = player.getDuration();
              if (duration) {
                  var value = (currentTime / duration) * 100;
                  progressBar.style.width = value + '%';
              }
          }
          function updateCurrentTimeDisplay() {
              var currentTimeDisplay = document.getElementById('current-time');
              var currentTime = player.getCurrentTime();
              var minutes = Math.floor(currentTime / 60).toString();
              var seconds = Math.floor(currentTime % 60).toString().padStart(2, '0');
              currentTimeDisplay.innerHTML = minutes + ':' + seconds;
          }
          function togglePlayPauseButton(isPlaying) {
              var playButton = document.getElementById('play');
              var pauseButton = document.getElementById('pause');
              if (isPlaying) {
                  playButton.classList.add('hide');
                  pauseButton.classList.remove('hide');
              } else {
                  playButton.classList.remove('hide');
                  pauseButton.classList.add('hide');
              }
          }
          function toggleFullScreen() {
              var videoContainer = document.querySelector('.px-video-container');
              if (!document.fullscreenElement) {
                  if (videoContainer.requestFullscreen) {
                      videoContainer.requestFullscreen();
                  } else if (videoContainer.mozRequestFullScreen) { /* Firefox */
                      videoContainer.mozRequestFullScreen();
                  } else if (videoContainer.webkitRequestFullscreen) { /* Chrome, Safari and Opera */
                      videoContainer.webkitRequestFullscreen();
                  } else if (videoContainer.msRequestFullscreen) { /* IE/Edge */
                      videoContainer.msRequestFullscreen();
                  }
              } else {
                  if (document.exitFullscreen) {
                      document.exitFullscreen();
                  } else if (document.mozCancelFullScreen) { /* Firefox */
                      document.mozCancelFullScreen();
                  } else if (document.webkitExitFullscreen) { /* Chrome, Safari and Opera */
                      document.webkitExitFullscreen();
                  } else if (document.msExitFullscreen) { /* IE/Edge */
                      document.msExitFullscreen();
                  }
              }
        }
